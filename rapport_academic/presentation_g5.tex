    \begin{columns}
        \begin{column}{0.48\textwidth}
            \begin{block}{\small Architecture Double-Bucket}
                \scriptsize
                \begin{itemize}
                    \item \textbf{Isolation CDNU} : 22 Buckets provisionnés (1 Principal + 1 Logs d'audit pour chacun des 11 CDNUs).
                    \item \textbf{Sécurisation Stricte} : Block Public Access complet et versioning actif pour contrer les ransomwares.

                \end{itemize}
            \end{block}
        \end{column}
        \begin{column}{0.52\textwidth}
            \begin{lstlisting}[basicstyle=\tiny\ttfamily,language=bash,caption={\tiny Bucket Policy HTTPS Forcé}]
data "aws_iam_policy_document" "s3_policy" {
  statement {
    sid    = "DenyInsecureTransport"
    effect = "Deny"
    principals { type = "*" \  identifiers = ["*"] }
    actions   = ["s3:*"]
    resources = [aws_s3_bucket.main.arn, "${aws_s3_bucket.main.arn}/*"]
    condition {
      test     = "Bool"
      variable = "aws:SecureTransport"
      values   = ["false"]
    }
  }
}
            \end{lstlisting}
        \end{column}
    \end{columns}
