% Slide 12 : RDS PostgreSQL - Stockage GP3 \& Performances
\begin{frame}{RDS PostgreSQL - Stockage GP3 \& Performances}
    \begin{itemize}
        \item \textbf{Type de disque} : General Purpose SSD de type **GP3**.
        \item \textbf{Garanties de performances} :
        \begin{itemize}
            \item **3000 IOPS** garantis de base sans frais supplémentaires.
            \item Débit stable de **125 Mo/s** linéaire.
            \item Permet d'isoler les performances d'I/O de la capacité globale du disque (contrairement aux volumes GP2 dépendants de la taille).
        \end{itemize}
        \item \textbf{Avantage} : Excellent temps de réponse lors des pics d'accès.
    \end{itemize}
\end{frame}

% Slide 13 : RDS PostgreSQL - Sécurité et Isolation Réseau
\begin{frame}{RDS PostgreSQL - Sécurité et Isolation Réseau}
    \begin{block}{Isolation stricte au niveau Réseau}
        \begin{itemize}
            \item \textbf{Subnet Group Privé} : Déploiement exclusif dans des sous-réseaux privés internes sans point d'accès ou d'IP publique Internet.
            \item \textbf{Sécurisation par Groupe de Sécurité} : 
            \begin{itemize}
                \item Restriction du protocole TCP sur le port **5432**.
                \item Seuls les flux provenant des groupes de sécurité de l'API ECS Fargate (Groupe 6) ou du bastion réseau (Groupe 4) sont acceptés.
                \item Tout autre trafic interne ou externe est systématiquement bloqué.
            \end{itemize}
        \end{itemize}
    \end{block}
\end{frame}

% Slide 14 : RDS PostgreSQL - Chiffrement et Souveraineté (KMS)
\begin{frame}{RDS PostgreSQL - Chiffrement et Souveraineté (KMS)}
    \begin{itemize}
        \item \textbf{Chiffrement au repos (At Rest)} : Toutes les tables SQL, index, fichiers de transactions de données et snapshots de sauvegarde sont chiffrés.
        \item \textbf{Algorithme appliqué} : Standard industriel robuste **AES-256**.
        \item \textbf{Clé KMS Dédiée} : Création et gestion autonome d'une clé KMS de chiffrement souveraine pour le projet national, assurant que même AWS ne peut lire les données stockées physiques sans autorisation.
    \end{itemize}
\end{frame}

% Slide 15 : AWS Secrets Manager - Gestion des Mots de Passe
\begin{frame}{AWS Secrets Manager - Gestion des Mots de Passe}
    \begin{block}{Cinématique de Sécurisation des Identifiants}
        \begin{itemize}
            \item \textbf{Génération Aléatoire IaC} : Terraform génère un mot de passe d'administration robuste lors du premier déploiement sans l'exposer à l'écran.
            \item \textbf{Stockage Chiffré} : Le mot de passe est directement écrit dans **AWS Secrets Manager** sous clé KMS.
            \item \textbf{Consommation Applicative} : L'API ECS (Groupe 6) interroge l'API Secrets Manager via son rôle d'exécution IAM pour lire le secret à la volée, évitant tout mot de passe en clair dans le code.
        \end{itemize}
    \end{block}
\end{frame}

% Slide 20 : Code IaC - Forcer HTTPS sur S3 via Bucket Policy
\begin{frame}[fragile]{Code IaC - Forcer HTTPS sur S3 via Bucket Policy}

data "aws_iam_policy_document" "bucket_policy" {
  statement {
    sid    = "DenyInsecureTransport"
    effect = "Deny"
    principals {
      type        = "*"
      identifiers = ["*"]
    }
    actions = ["s3:*"]
    resources = [
      aws_s3_bucket.main.arn,
      "${aws_s3_bucket.main.arn}/*"
    ]
    condition {
      test     = "Bool"
      variable = "aws:SecureTransport"
      values   = ["false"]
    }
  }
}
    \end{lstlisting}

% Slide 23 : S3 Buckets - Sécurisation des Accès Réseau
\begin{frame}{S3 Buckets - Sécurisation des Accès Réseau}
    \begin{block}{Contrôles de Sécurité Rigoureux}
        \begin{itemize}
            \item \textbf{Block Public Access} : Option activée globalement empêchant l'exposition publique accidentelle de fichiers.
            \item \textbf{Versioning Actif} : Sauvegarde automatique de l'historique des fichiers en cas d'altération ou d'attaque par ransomware.
            \item \textbf{Bucket Policy d'Exclusion HTTP} : Exigence absolue du transit sécurisé HTTPS TLS 1.2+. Les requêtes non chiffrées sont rejetées à la source.
        \end{itemize}
    \end{block}
\end{frame}

% Slide 24 : S3 Buckets - Configuration CORS \& Upload Direct

    \begin{itemize}

        \item \textbf{Solution CORS intégrée} : Le module configure des règles Cross-Origin Resource Sharing (CORS) autorisées pour les domaines universitaires nationaux.
        \item \textbf{Mécanisme Direct Upload} : Les navigateurs des étudiants peuvent téléverser directement des fichiers volumineux de manière sécurisée vers S3 après obtention d'une URL pré-signée de l'API.

% Slide 25 : S3 Buckets - Stratégie FinOps \& Cycle de Vie
\begin{frame}{S3 Buckets - Stratégie FinOps \& Cycle de Vie}
    \begin{block}{Politique de cycle de vie (Lifecycle Rules)}
        \begin{itemize}
            \item \textbf{0 à 30 jours (S3 Standard)} : Données actives fréquemment consultées.
            \item \textbf{31 à 120 jours (Glacier Instant Retrieval)} : Passage des données tièdes avec accès immédiat en millisecondes (**80\% d'économie**).
            \item \textbf{121 à 210 jours (Glacier Flexible Retrieval)} : Archivage avec récupération en 3 à 5 heures (**90\% d'économie**).
            \item \textbf{Après 210 jours (Glacier Deep Archive)} : Archivage de longue durée avec récupération en 12 heures (**95\% d'économie**).
        \end{itemize}
    \end{block}
\end{frame}

\begin{frame}{AWS Backup - Politique Nationale de Sauvegarde}
    \begin{block}{Règles de clichés inaltérables}
        \begin{itemize}
            \item \textbf{Sauvegarde Quotidienne} : Snapshot automatique déclenché à 03h00 (UTC), conservé **7 jours** dans le coffre.
            \item \textbf{Sauvegarde Hebdomadaire} : Snapshot complet déclenché le dimanche à 04h00 (UTC), conservé **4 semaines** (28 jours).
